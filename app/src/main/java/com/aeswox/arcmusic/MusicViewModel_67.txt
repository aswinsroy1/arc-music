604:     private fun computeListeningStats(
605:         history: List<PlayHistory>,
606:         tracks: List<Track>
607:     ): ListeningStatsData {
608:         if (history.isEmpty() || tracks.isEmpty()) {
609:             return ListeningStatsData(
610:                 totalMinutes = 0,
611:                 weekOverWeekPct = null,
612:                 weeklyMinutesByDay = List(7) { 0L },
613:                 topArtists = emptyList(),
614:                 topGenres = computeTopGenresByCount(tracks),
615:                 nightOwlMinutesByHour = List(24) { 0L }
616:             )
617:         }
618: 
619:         val trackById: Map<String, Track> = tracks.associateBy { it.id }
620: 
621:         // --- Total listening time (actual played ms, not full duration) ---
622:         val totalMs = history.sumOf { ph -> getEffectivePlayedMs(ph, trackById[ph.trackId]) }
623:         val totalMinutes = totalMs / 60_000L
624: 
625:         // --- Week-over-week trend ---
626:         val now = System.currentTimeMillis()
627:         val thisWeekStart = now - 7L * 24 * 3600 * 1000
628:         val prevWeekStart = now - 14L * 24 * 3600 * 1000
629: 
630:         val thisWeekMs = history
631:             .filter { it.timestamp >= thisWeekStart }
632:             .sumOf { ph -> getEffectivePlayedMs(ph, trackById[ph.trackId]) }
633: 
634:         // Only show the trend if we have actual history in the *prior* week
635:         val prevWeekHistory = history.filter { it.timestamp in prevWeekStart until thisWeekStart }
636:         val weekOverWeekPct: Int? = if (prevWeekHistory.isEmpty()) {
637:             null // Not enough history yet — hide the line
638:         } else {
639:             val prevWeekMs = prevWeekHistory.sumOf { ph -> getEffectivePlayedMs(ph, trackById[ph.trackId]) }
640:             if (prevWeekMs == 0L) null
641:             else ((thisWeekMs - prevWeekMs) * 100L / prevWeekMs).toInt()
642:         }
643: 
644:         // --- Weekly activity: minutes per day of current 7-day window (Mon..Sun) ---
645:         val cal = java.util.Calendar.getInstance()
646:         cal.timeInMillis = now
647:         // Roll back to start of today
648:         cal.set(java.util.Calendar.HOUR_OF_DAY, 0)
649:         cal.set(java.util.Calendar.MINUTE, 0)
650:         cal.set(java.util.Calendar.SECOND, 0)
651:         cal.set(java.util.Calendar.MILLISECOND, 0)
652:         val dayMs = 24L * 3600 * 1000
653: 
654:         // Build 7 buckets: index 0 = 6 days ago, index 6 = today
655:         val weeklyMinutesByDay = (6 downTo 0).map { daysAgo ->
656:             val dayStart = cal.timeInMillis - daysAgo * dayMs
657:             val dayEnd = dayStart + dayMs
658:             history
659:                 .filter { it.timestamp in dayStart until dayEnd }
660:                 .sumOf { ph -> getEffectivePlayedMs(ph, trackById[ph.trackId]) } / 60_000L
661:         }
662: 
663:         // --- Top Artists (by approximate listening time) ---
664:         val artistMinutes = mutableMapOf<String, Long>()
665:         history.forEach { ph ->
666:             val track = trackById[ph.trackId] ?: return@forEach
667:             val artist = track.artist.ifBlank { return@forEach }
668:             val playedMin = getEffectivePlayedMs(ph, track) / 60_000L
669:             artistMinutes[artist] = (artistMinutes[artist] ?: 0L) + playedMin
670:         }
671:         // Build artist entries — photoUri comes from the Artists table via libraryArtists,
672:         // but since we only have Track here, we leave photoUri null (placeholder shown in UI).
673:         val topArtists = artistMinutes.entries
674:             .sortedByDescending { it.value }
675:             .take(8)
676:             .map { (name, minutes) -> ArtistStatEntry(name, null, minutes) }
677: 
678:         // --- Top Genres (ranked by actual listening time) ---
679:         val topGenres = computeTopGenresByListeningTime(history, trackById)
680: 
681:         // --- Listening Personality: computed from any available history ---
682:         val nightOwlData: List<Long> = run {
683:             val minutesByHour = LongArray(24)
684:             history.forEach { ph ->
685:                 val track = trackById[ph.trackId]
686:                 val durationMin = getEffectivePlayedMs(ph, track) / 60_000L
687:                 val hourCal = java.util.Calendar.getInstance()
688:                 hourCal.timeInMillis = ph.timestamp
689:                 val hour = hourCal.get(java.util.Calendar.HOUR_OF_DAY)
690:                 minutesByHour[hour] += durationMin
691:             }
692:             minutesByHour.toList()
693:         }
694: 
695:         return ListeningStatsData(
696:             totalMinutes = totalMinutes,
697:             weekOverWeekPct = weekOverWeekPct,
698:             weeklyMinutesByDay = weeklyMinutesByDay,
699:             topArtists = topArtists,
700:             topGenres = topGenres,
701:             nightOwlMinutesByHour = nightOwlData
702:         )
703:     }
